cmake_minimum_required(VERSION 3.16)
project(ytcui_dl VERSION 0.6.1 LANGUAGES CXX)
if(WIN32)
  message(FATAL_ERROR "ytcui-dl uses POSIX transport/session APIs. Use WSL on Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(YTDL_WERROR "Treat warnings as errors" OFF)
option(YTDL_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
find_package(OpenSSL REQUIRED)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
add_library(ytcui_dependencies INTERFACE)
target_include_directories(ytcui_dependencies INTERFACE include)
target_link_libraries(ytcui_dependencies INTERFACE OpenSSL::SSL OpenSSL::Crypto ZLIB::ZLIB Threads::Threads)
target_compile_definitions(ytcui_dependencies INTERFACE YTFAST_VERSION="${PROJECT_VERSION}")
target_compile_options(ytcui_dependencies INTERFACE -Wall -Wextra)
if(YTDL_WERROR)
  target_compile_options(ytcui_dependencies INTERFACE -Werror)
endif()
if(YTDL_SANITIZERS)
  target_compile_options(ytcui_dependencies INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ytcui_dependencies INTERFACE -fsanitize=address,undefined)
endif()
if(APPLE)
  target_compile_definitions(ytcui_dependencies INTERFACE YTFAST_MACOS)
elseif(CMAKE_SYSTEM_NAME MATCHES "FreeBSD|OpenBSD|NetBSD")
  target_compile_definitions(ytcui_dependencies INTERFACE YTFAST_BSD)
  set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)
endif()
add_executable(ytcui-dl cli/ytcui-dl.cpp)
target_link_libraries(ytcui-dl PRIVATE ytcui_dependencies)
include(CTest)
if(BUILD_TESTING)
  foreach(name yj client select sqlite audit)
    add_executable(test_${name} test/test_${name}.cpp)
    target_link_libraries(test_${name} PRIVATE ytcui_dependencies)
    add_test(NAME ${name} COMMAND test_${name} ${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures)
    set_tests_properties(${name} PROPERTIES TIMEOUT 120)
  endforeach()
  add_executable(test_transfer test/test_transfer.cpp)
  target_link_libraries(test_transfer PRIVATE ytcui_dependencies)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_cli.py $<TARGET_FILE:ytcui-dl>)
  add_test(NAME transfers COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_transfers.py $<TARGET_FILE:test_transfer>)
  set_tests_properties(cli transfers PROPERTIES TIMEOUT 180)
endif()
install(TARGETS ytcui-dl RUNTIME DESTINATION bin)
install(FILES README.md LICENSE AUDIT.md DESTINATION share/doc/ytcui-dl)
